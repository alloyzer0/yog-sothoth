function(ys_enable_strict_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endfunction()

add_executable(abi_c17_compile c17_compile.c)
target_link_libraries(abi_c17_compile PRIVATE yog_sothoth_runtime_headers)
target_compile_features(abi_c17_compile PRIVATE c_std_17)
set_target_properties(abi_c17_compile PROPERTIES C_EXTENSIONS OFF)
target_compile_definitions(abi_c17_compile PRIVATE YS_STATIC)
ys_enable_strict_warnings(abi_c17_compile)
add_test(NAME abi.c17.compile COMMAND abi_c17_compile)
